    suma = 0
    for x in tab:
        suma += x
    return suma / len(tab)

b) def indeks_min(tab):
    indeks = 0

    for i in range(1, len(tab)):
        if tab[i] < tab[indeks]:
            indeks = i

    return indeks
c) def ile_wiekszych(tab, wartosc):
    licznik = 0

    for x in tab:
        if x > wartosc:
            licznik += 1

    return licznik

def sprawdz(wyrazenie):
    pary = {")": "(", "]": "[", "}": "{"}
    stos = []

    for znak in wyrazenie:
        if znak in "([{":
            stos.append(znak)
        elif znak in pary:
            if not stos or stos.pop() != pary[znak]:
                return False

    return len(stos) == 0

print(sprawdz("( a + [ b * c ] )"))	# True
print(sprawdz("{ ( a + b ]  }"))	# False
print(sprawdz("( ( a + b )"))		# False
